// Package proposal — 工作流阶段与闸门
package proposal

const (
	StageParse    = "parse"
	StageGenerate = "generate"
	StageCheck    = "check"
	StageFormat   = "format"
)

var stageRank = map[string]int{StageParse: 1, StageGenerate: 2, StageCheck: 3, StageFormat: 4}

// requireStage 检查阶段：有招标文件但未解析时阻止生成类操作
func (s *Service) requireStage(p *Proposal, need string) error {
	if p.BidSummary != nil && len(p.BidSummary.RawFiles) > 0 {
		if stageRank[p.Stage] < stageRank[need] {
			return errStageGate(need)
		}
	}
	return nil
}

// advanceStage 推进到更高阶段（幂等）
func (p *Proposal) advanceStage(next string) {
	if stageRank[next] > stageRank[p.Stage] {
		p.Stage = next
	}
}

func errStageGate(need string) error {
	switch need {
	case StageGenerate:
		return fmt.Errorf("阶段闸门：请先完成招标解析（AI 分析）后再生成大纲/章节")
	default:
		return fmt.Errorf("阶段闸门：请先完成上一阶段（需要 %s）", need)
	}
}
```

`types.go` Proposal 增加 `Stage string \`json:"stage,omitempty"\``。

- [ ] **Step 4: 接线闸门与推进**

- `outline_gen.go` GenerateOutline：`s.requireStage(p, StageGenerate)`（store.Get 后）；成功后 `p.advanceStage(StageGenerate)`。
- `context.go` SectionContext：开头 store.Get 后 `s.requireStage(p, StageGenerate)`（GenerateSection/流式共用）。
- `batch.go` RunBatch：加载 p 后 `s.requireStage(p, StageGenerate)`。
- `parse.go` parseBidFile：成功后 `p.advanceStage(StageParse)`。
- `service.go` CheckAll：完成后 `p.advanceStage(StageCheck)` 并 store.Update；ExportDocxWithOptions/ExportSectionDocx：成功导出后 `p.advanceStage(StageFormat)` + store.Update。

注意：`requireStage` 只对“有 RawFiles 且 stage 不足”拦截；空白方案（无招标文件）不拦截，保持既有测试通过。`stage_test` 中 GenerateSection 传入不存在章节 id 会先触发闸门错误（顺序：SectionContext 先 requireStage 再查章节），符合断言。

// Package proposal — 一键流水线
package proposal

import (
	"context"
	"fmt"
)

// PipelineProgress 流水线进度回调
type PipelineProgress func(step, status, detail string)

// RunPipeline 按四阶段顺序执行可运行步骤：解析→大纲→批量生成→全面检查
func (s *Service) RunPipeline(ctx context.Context, proposalID string, onProgress PipelineProgress) (*Proposal, []CheckItem, error) {
	p, err := s.store.Get(proposalID)
	if err != nil {
		return nil, nil, err
	}
	if onProgress != nil {
		onProgress("parse", "start", "")
	}
	if p.BidSummary != nil && len(p.BidSummary.RawFiles) > 0 && p.Stage == "" {
		if _, err := s.ParseBidFileWithProgress(ctx, proposalID, func(stage, detail string) {
			if onProgress != nil {
				onProgress("parse", stage, detail)
			}
		}); err != nil {
			return nil, nil, fmt.Errorf("招标解析失败: %w", err)
		}
	}
	if onProgress != nil {
		onProgress("outline", "start", "")
	}
	p, _ = s.store.Get(proposalID)
	if len(p.Sections) == 0 {
		if _, err := s.GenerateOutline(ctx, proposalID, p.Requirements, "", 0); err != nil {
			return nil, nil, fmt.Errorf("大纲生成失败: %w", err)
		}
	}
	if onProgress != nil {
		onProgress("generate", "start", "")
	}
	p, _ = s.store.Get(proposalID)
	if err := s.RunBatch(ctx, proposalID, func(cur, total int, sid, status string, words int) {
		if onProgress != nil {
			onProgress("generate", status, fmt.Sprintf("%d/%d", cur, total))
		}
	}); err != nil {
		return nil, nil, err
	}
	if onProgress != nil {
		onProgress("check", "start", "")
	}
	p, items, err := s.CheckAll(ctx, proposalID)
	if err != nil {
		return nil, nil, err
	}
	if onProgress != nil {
		onProgress("check", "done", fmt.Sprintf("%d 项", len(items)))
	}
	return p, items, nil
}
```

func (a *officeState) ProposalRunPipeline(pid string) {
	if a.proposalSvc == nil {
		return
	}
	go func() {
		defer func() { if r := recover(); r != nil { slog.Error("office: pipeline panic recovered", "panic", r) } }()
		_, items, err := a.proposalSvc.RunPipeline(a.ctx, pid, func(step, status, detail string) {
			a.emit("proposal-pipeline-progress", map[string]interface{}{"step": step, "status": status, "detail": detail})
		})
		if err != nil {
			a.emit("proposal-pipeline-progress", map[string]interface{}{"step": "error", "status": "error", "detail": err.Error()})
			return
		}
		a.emit("proposal-pipeline-progress", map[string]interface{}{"step": "done", "status": "done", "detail": fmt.Sprintf("检查 %d 项", len(items))})
	}()
}
